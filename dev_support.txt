# 1. Create the new XML-based solution file
dotnet new sln --name ImpactSupport --format slnx

# 2. Create backend project with Controllers
dotnet new webapi `
    -n ImpactSupport.Api `
    -o backend/ImpactSupport.Api `
    --use-controllers

# 3. Add backend project to root solution
dotnet sln ImpactSupport.slnx add backend/ImpactSupport.Api/ImpactSupport.Api.csproj

# 4. Verify solution
dotnet sln ImpactSupport.slnx list

# 5. Build entire solution
dotnet build ImpactSupport.slnx
